import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int l = sc.nextInt();
        int r = sc.nextInt();
        int a = sc.nextInt();

        int min = Math.min(l, r);
        int diff = Math.abs(l - r);

        int result = 2 * (min + Math.min(a, diff) + (a - Math.min(a, diff)) / 2);

        System.out.println(result);
    }
}
