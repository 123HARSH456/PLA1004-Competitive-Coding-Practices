import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            int[] a = new int[7];
            int total = 0;

            for (int i = 0; i < 7; i++) {
                a[i] = sc.nextInt();
                total += a[i];
            }

            int ans = Integer.MIN_VALUE;

            for (int i = 0; i < 7; i++) {
                ans = Math.max(ans, 2 * a[i] - total);
            }

            System.out.println(ans);
        }

        sc.close();
    }
}
