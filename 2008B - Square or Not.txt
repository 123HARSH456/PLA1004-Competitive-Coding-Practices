import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();
            String s = sc.next();

            int x = (int) Math.sqrt(n);

            // n must be a perfect square
            if (x * x != n) {
                System.out.println("No");
                continue;
            }

            boolean valid = true;

            for (int i = 0; i < n; i++) {
                int row = i / x;
                int col = i % x;

                boolean edge = (row == 0 || row == x - 1 ||
                                col == 0 || col == x - 1);

                if (edge && s.charAt(i) != '1') {
                    valid = false;
                    break;
                }

                if (!edge && s.charAt(i) != '0') {
                    valid = false;
                    break;
                }
            }

            System.out.println(valid ? "Yes" : "No");
        }
    }
}
