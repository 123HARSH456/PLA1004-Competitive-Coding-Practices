import java.util.*;

public class Main {

    static boolean isVowel(char c) {
        return c == 'a' || c == 'e' || c == 'i' ||
               c == 'o' || c == 'u' || c == 'y';
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        String s = sc.next();

        StringBuilder ans = new StringBuilder();

        for (int i = 0; i < n; i++) {
            char curr = s.charAt(i);

            // Skip curr if both previous kept char and curr are vowels
            if (ans.length() > 0 &&
                isVowel(ans.charAt(ans.length() - 1)) &&
                isVowel(curr)) {
                continue;
            }

            ans.append(curr);
        }

        System.out.println(ans);
    }
}
