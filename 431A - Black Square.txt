import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int[] a = new int[4];

        for (int i = 0; i < 4; i++) {
            a[i] = sc.nextInt();
        }

        String s = sc.next();

        int calories = 0;

        for (char c : s.toCharArray()) {
            int strip = c - '1';  // '1' -> 0, '2' -> 1, etc.
            calories += a[strip];
        }

        System.out.println(calories);
    }
}
