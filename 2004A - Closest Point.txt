import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();
            int[] x = new int[n];

            for (int i = 0; i < n; i++) {
                x[i] = sc.nextInt();
            }

            int[] nearest = new int[n];

            for (int i = 0; i < n; i++) {
                if (i == 0) {
                    nearest[i] = x[1] - x[0];
                } else if (i == n - 1) {
                    nearest[i] = x[n - 1] - x[n - 2];
                } else {
                    nearest[i] = Math.min(
                        x[i] - x[i - 1],
                        x[i + 1] - x[i]
                    );
                }
            }

            boolean possible = false;

            for (int p = 0; p <= 200; p++) {
                boolean ok = true;

                for (int i = 0; i < n; i++) {
                    if (p == x[i] || Math.abs(p - x[i]) > nearest[i]) {
                        ok = false;
                        break;
                    }
                }

                if (ok) {
                    possible = true;
                    break;
                }
            }

            System.out.println(possible ? "YES" : "NO");
        }

        sc.close();
    }
}
