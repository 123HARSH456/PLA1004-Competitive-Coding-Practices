import java.util.*;

public class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while(t-- > 0){

            long n = sc.nextLong();
            long m = sc.nextLong();
            long r = sc.nextLong();
            long c = sc.nextLong();

            long i = (r-1)*m + c;

            long ans = n*m - i + (n-r)*(m-1);

            System.out.println(ans);
        }
    }
}
