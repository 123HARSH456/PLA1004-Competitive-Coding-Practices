import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();
            int[] v = new int[n];

            for (int i = 0; i < n; i++) {
                v[i] = sc.nextInt();
            }

            for (int i = 0, j = n - 1; i < n / 2; i++, j--) {
                System.out.print(v[i] + " " + v[j] + " ");
            }

            if (n % 2 != 0) {
                System.out.print(v[n / 2]);
            }

            System.out.println();
        }

        sc.close();
    }
}
